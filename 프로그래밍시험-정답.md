# [문항1] 학생 관리 시스템 — 정답 (Java)

### Student.java

```java
public class Student {
    String name;
    int age;

    public Student(String name, int age) {
        this.name = name;
        this.age = age;
    }

    @Override
    public String toString() {
        return "Student{name='" + name + "', age=" + age + "}";
    }
}
```

### StudentApp.java

```java
import java.util.ArrayList;
import java.util.Scanner;

public class StudentApp {
    public static void main(String[] args) {
        ArrayList<Student> list = new ArrayList<>();
        Scanner sc = new Scanner(System.in);
        boolean run = true;

        while (run) {
            System.out.println("------------------------");
            System.out.println("1. 학생 등록");
            System.out.println("2. 학생 검색");
            System.out.println("3. 학생 수정");
            System.out.println("4. 학생 삭제");
            System.out.println("5. 전체 출력");
            System.out.println("6. 종료");
            System.out.print("선택 > ");

            int menu;
            try {
                menu = Integer.parseInt(sc.nextLine());
            } catch (Exception e) {
                System.out.println("숫자만 입력하세요");
                continue;
            }

            switch (menu) {
                case 1:
                    System.out.print("이름 입력 : ");
                    String name = sc.nextLine();
                    int age;
                    while (true) {
                        try {
                            System.out.print("나이 입력 : ");
                            age = Integer.parseInt(sc.nextLine());
                            break;
                        } catch (Exception e) {
                            System.out.println("나이는 숫자로 입력하세요");
                        }
                    }
                    list.add(new Student(name, age));
                    System.out.println("학생 등록 완료");
                    break;

                case 2:
                    System.out.print("검색할 이름 : ");
                    name = sc.nextLine();
                    Student s = findStudent(list, name);
                    if (s != null) System.out.println(s);
                    else System.out.println("학생이 없습니다");
                    break;

                case 3:
                    System.out.print("수정할 이름 : ");
                    name = sc.nextLine();
                    int newAge;
                    while (true) {
                        try {
                            System.out.print("새 나이 입력 : ");
                            newAge = Integer.parseInt(sc.nextLine());
                            break;
                        } catch (Exception e) {
                            System.out.println("나이는 숫자로 입력하세요");
                        }
                    }
                    if (updateStudent(list, name, newAge))
                        System.out.println("수정 완료");
                    else
                        System.out.println("해당 학생이 없습니다");
                    break;

                case 4:
                    System.out.print("삭제할 이름 : ");
                    name = sc.nextLine();
                    if (deleteStudent(list, name))
                        System.out.println("삭제 완료");
                    else
                        System.out.println("삭제할 학생이 없습니다");
                    break;

                case 5:
                    if (list.isEmpty())
                        System.out.println("등록된 학생이 없습니다");
                    else
                        for (Student student : list)
                            System.out.println(student);
                    break;

                case 6:
                    run = false;
                    System.out.println("프로그램 종료");
                    break;

                default:
                    System.out.println("1~6번 중에 선택하세요");
            }
        }
        sc.close();
    }

    static Student findStudent(ArrayList<Student> list, String name) {
        for (Student s : list) {
            if (s.name.equals(name)) return s;
        }
        return null;
    }

    static boolean updateStudent(ArrayList<Student> list, String name, int newAge) {
        for (Student s : list) {
            if (s.name.equals(name)) {
                s.age = newAge;
                return true;
            }
        }
        return false;
    }

    static boolean deleteStudent(ArrayList<Student> list, String name) {
        for (int i = 0; i < list.size(); i++) {
            if (list.get(i).name.equals(name)) {
                list.remove(i);
                return true;
            }
        }
        return false;
    }
}
```

# [문항2] Product 클래스 — 정답 (Java)

### Product.java

```java
public class Product {
    String name;
    int price;
    int quantity;

    public Product() {
    }

    public Product(String name, int price, int quantity) {
        this.name = name;
        this.price = price;
        this.quantity = quantity;
    }

    @Override
    public String toString() {
        return "상품명: " + name + ", 가격: " + price + "원, 수량: " + quantity + "개";
    }

    public int getTotalPrice() {
        return price * quantity;
    }
}
```

### Problem2.java (실행·테스트용)

```java
public class Problem2 {
    public static void main(String[] args) {
        Product p1 = new Product("노트북", 1000000, 2);
        Product p2 = new Product("마우스", 15000, 5);

        System.out.println(p1);
        System.out.println("총 가격: " + p1.getTotalPrice() + "원");
        System.out.println(p2);
        System.out.println("총 가격: " + p2.getTotalPrice() + "원");
    }
}
```

# [문항3] NumberManager — 정답 (Java)

### NumberManager.java

```java
import java.util.ArrayList;

public class NumberManager {

    public static void addNumber(ArrayList list, int num) {
        list.add(num);
    }

    public static int findMax(ArrayList list) {
        if (list == null || list.isEmpty()) return -1;
        int max = (Integer) list.get(0);
        for (int i = 1; i < list.size(); i++) {
            int v = (Integer) list.get(i);
            if (v > max) max = v;
        }
        return max;
    }

    public static int findMin(ArrayList list) {
        if (list == null || list.isEmpty()) return -1;
        int min = (Integer) list.get(0);
        for (int i = 1; i < list.size(); i++) {
            int v = (Integer) list.get(i);
            if (v < min) min = v;
        }
        return min;
    }

    public static int getSum(ArrayList list) {
        int sum = 0;
        for (Object o : list) {
            sum += (Integer) o;
        }
        return sum;
    }
}
```

# [문항4] StringUtil — 정답 (Java)

### StringUtil.java

```java
public class StringUtil {

    public static int countChar(String text, char ch) {
        if (text == null) return 0;
        int count = 0;
        for (int i = 0; i < text.length(); i++) {
            if (text.charAt(i) == ch) count++;
        }
        return count;
    }

    public static String reverseString(String text) {
        if (text == null) return null;
        StringBuilder sb = new StringBuilder(text);
        return sb.reverse().toString();
    }

    public static String toUpperCase(String text) {
        if (text == null) return null;
        return text.toUpperCase();
    }

    public static String removeSpaces(String text) {
        if (text == null) return null;
        return text.replace(" ", "");
    }
}
```

`reverseString` 반복문 버전:

```java
public static String reverseString(String text) {
    if (text == null) return null;
    String result = "";
    for (int i = text.length() - 1; i >= 0; i--) {
        result += text.charAt(i);
    }
    return result;
}
```

# [문항5] C 최댓값·최솟값 — 정답

### problem5.c (또는 c_maxmin.c)

```c
#include <stdio.h>

int main() {
    int arr[5];
    int i, max, min;

    for (i = 0; i < 5; i++) {
        scanf("%d", &arr[i]);
    }

    max = arr[0];
    min = arr[0];

    for (i = 1; i < 5; i++) {
        if (arr[i] > max) max = arr[i];
        if (arr[i] < min) min = arr[i];
    }

    printf("최댓값: %d\n", max);
    printf("최솟값: %d\n", min);
    return 0;
}
```

# [문항6] C 평균 이상 개수 — 정답

### problem6.c

```c
#include <stdio.h>

int main() {
    int arr[10];
    int i, sum = 0, count = 0;
    float avg;

    for (i = 0; i < 10; i++) {
        scanf("%d", &arr[i]);
        sum += arr[i];
    }

    avg = (float)sum / 10;

    for (i = 0; i < 10; i++) {
        if (arr[i] >= avg) count++;
    }

    printf("평균: %.2f\n", avg);
    printf("평균 이상인 값의 개수: %d\n", count);
    return 0;
}
```

# [문항7] C 문자 개수 — 정답

### problem7.c

```c
#include <stdio.h>
#include <string.h>

int main() {
    char str[101];
    char ch;
    int i, len, count = 0;

    fgets(str, 101, stdin);
    len = strlen(str);
    if (len > 0 && str[len - 1] == '\n') {
        str[len - 1] = '\0';
        len--;
    }

    scanf(" %c", &ch);

    for (i = 0; i < len; i++) {
        if (str[i] == ch) count++;
    }

    printf("%c가 나타나는 횟수: %d\n", ch, count);
    return 0;
}
```

# [문항8] C 짝수·홀수 분리 — 정답

### problem8.c

```c
#include <stdio.h>

int main() {
    int arr[8];
    int even[8], odd[8];
    int i, e = 0, o = 0;

    for (i = 0; i < 8; i++) {
        scanf("%d", &arr[i]);
        if (arr[i] % 2 == 0)
            even[e++] = arr[i];
        else
            odd[o++] = arr[i];
    }

    printf("짝수: ");
    if (e == 0) {
        printf("없음\n");
    } else {
        for (i = 0; i < e; i++) {
            printf("%d", even[i]);
            if (i < e - 1) printf(" ");
        }
        printf("\n");
    }

    printf("홀수: ");
    if (o == 0) {
        printf("없음\n");
    } else {
        for (i = 0; i < o; i++) {
            printf("%d", odd[i]);
            if (i < o - 1) printf(" ");
        }
        printf("\n");
    }

    return 0;
}
```
