# 5주차 정답

| 번호 | 정답 |
|------|------|
| 5-1 | ② 다음 case도 실행됨 (fall-through) |
| 5-2 | `B` |
| 5-3 | ③ float (실수는 switch 불가) |
| 5-4 | age≥18 **그리고** score≥60 |

### 5-5

```c
#include <stdio.h>
int main() {
    int n;
    scanf("%d", &n);
    if (n > 0) printf("양수\n");
    else if (n < 0) printf("음수\n");
    else printf("0\n");
    return 0;
}
```

### 5-6

```c
#include <stdio.h>
int main() {
    int score;
    scanf("%d", &score);
    if (score >= 90) printf("A\n");
    else if (score >= 80) printf("B\n");
    else if (score >= 70) printf("C\n");
    else if (score >= 60) printf("D\n");
    else printf("F\n");
    return 0;
}
```

### 5-7

```c
#include <stdio.h>
int main() {
    int menu;
    scanf("%d", &menu);
    switch (menu) {
        case 1: printf("커피\n"); break;
        case 2: printf("차\n"); break;
        case 3: printf("주스\n"); break;
        default: printf("잘못된 선택\n");
    }
    return 0;
}
```

### 5-8

```c
#include <stdio.h>
int main() {
    int age, score;
    scanf("%d %d", &age, &score);
    if (age >= 18 && score >= 60) printf("합격\n");
    else printf("불합격\n");
    return 0;
}
```

# 6주차 정답

| 번호 | 정답 |
|------|------|
| 6-1 | 10번 |
| 6-2 | ② do-while은 최소 1번 실행 |
| 6-3 | `1 2 3 4 ` (5에서 break) |
| 6-4 | 현재 반복을 건너뛰고 다음 반복으로 |

### 6-9

```c
#include <stdio.h>
int main() {
    int i;
    for (i = 1; i <= 10; i++) printf("%d ", i);
    printf("\n");
    return 0;
}
```

### 6-10

```c
#include <stdio.h>
int main() {
    int dan, i;
    scanf("%d", &dan);
    for (i = 1; i <= 9; i++)
        printf("%d x %d = %d\n", dan, i, dan * i);
    return 0;
}
```

### 6-11

```c
#include <stdio.h>
int main() {
    int n, i, sum = 0;
    scanf("%d", &n);
    for (i = 1; i <= n; i++) sum += i;
    printf("합: %d\n", sum);
    return 0;
}
```

### 6-12

```c
#include <stdio.h>
int main() {
    int i, sum = 0;
    for (i = 2; i <= 100; i += 2) sum += i;
    printf("짝수의 합: %d\n", sum);
    return 0;
}
```

### 6-13

```c
#include <stdio.h>
int main() {
    int n, i, fact = 1;
    scanf("%d", &n);
    for (i = 1; i <= n; i++) fact *= i;
    printf("%d\n", fact);
    return 0;
}
```

### 6-14

```c
#include <stdio.h>
int main() {
    int i, j;
    for (i = 1; i <= 5; i++) {
        for (j = 1; j <= i; j++) printf("*");
        printf("\n");
    }
    return 0;
}
```

# 7주차 정답

| 번호 | 정답 |
|------|------|
| 7-1 | ① 0~4 |
| 7-2 | `30` |
| 7-3 | `arr[0]` (첫 번째 요소) |

### 7-15

```c
#include <stdio.h>
int main() {
    int arr[5], i;
    for (i = 0; i < 5; i++) scanf("%d", &arr[i]);
    for (i = 0; i < 5; i++) printf("%d ", arr[i]);
    printf("\n");
    return 0;
}
```

### 7-16

```c
#include <stdio.h>
int main() {
    int arr[5], i, sum = 0;
    for (i = 0; i < 5; i++) {
        scanf("%d", &arr[i]);
        sum += arr[i];
    }
    printf("합계: %d\n", sum);
    printf("평균: %.2f\n", (float)sum / 5);
    return 0;
}
```

### 7-17

```c
#include <stdio.h>
int main() {
    int arr[5], i, max, min;
    for (i = 0; i < 5; i++) scanf("%d", &arr[i]);
    max = min = arr[0];
    for (i = 1; i < 5; i++) {
        if (arr[i] > max) max = arr[i];
        if (arr[i] < min) min = arr[i];
    }
    printf("최댓값: %d\n", max);
    printf("최솟값: %d\n", min);
    return 0;
}
```

### 7-18

```c
#include <stdio.h>
int main() {
    int arr[5], i, search, found = 0;
    for (i = 0; i < 5; i++) scanf("%d", &arr[i]);
    scanf("%d", &search);
    for (i = 0; i < 5; i++) {
        if (arr[i] == search) {
            printf("%d는 %d번째 위치\n", search, i);
            found = 1;
            break;
        }
    }
    if (!found) printf("없음\n");
    return 0;
}
```

### 7-19

```c
#include <stdio.h>
int main() {
    int arr[5] = {1, 2, 3, 4, 5}, i;
    for (i = 4; i >= 0; i--) printf("%d ", arr[i]);
    printf("\n");
    return 0;
}
```

# 8주차 정답

| 번호 | 정답 |
|------|------|
| 8-1 | `\0` (널 문자) |
| 8-2 | 배열 이름은 **주소**이므로 `&` 불필요 |
| 8-3 | `0` (같음) |
| 8-4 | `strlen()` |

### 8-20

```c
#include <stdio.h>
#include <string.h>
int main() {
    char name[50];
    scanf("%s", name);
    printf("이름: %s\n", name);
    printf("길이: %d\n", (int)strlen(name));
    return 0;
}
```

### 8-21

```c
#include <stdio.h>
#include <string.h>
int main() {
    char str1[50], str2[50];
    scanf("%s %s", str1, str2);
    strcat(str1, " ");
    strcat(str1, str2);
    printf("%s\n", str1);
    return 0;
}
```

또는:

```c
strcat(str1, str2);  /* str2가 " World" 형태로 입력된 경우 */
```

### 8-22

```c
#include <stdio.h>
#include <string.h>
int main() {
    char str[100];
    char ch;
    int i, count = 0;
    scanf("%s", str);
    scanf(" %c", &ch);
    for (i = 0; i < strlen(str); i++)
        if (str[i] == ch) count++;
    printf("개수: %d\n", count);
    return 0;
}
```

### 8-23

```c
#include <stdio.h>
#include <string.h>
int main() {
    char a[50], b[50];
    scanf("%s %s", a, b);
    if (strcmp(a, b) == 0) printf("같음\n");
    else printf("다름\n");
    return 0;
}
```
