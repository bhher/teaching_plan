# 11주차 정답

| 번호 | 정답 |
|------|------|
| 11-1 | ② 서로 다른 자료형을 묶어 하나의 타입으로 만든다 |
| 11-2 | ② `s1.name` (구조체 변수 → `.` 연산자) |
| 11-3 | ② `p->name` (포인터 → `->` 연산자) |
| 11-4 | ② `Student s1;` 로 선언 가능 (`struct` 생략) |
| 11-5 | `홍길동 85` |
| 11-6 | ② 구조체의 **복사본**이 전달된다 (값 전달) |

### `.` vs `->` 정리

| 대상 | 연산자 | 예 |
|------|--------|-----|
| 구조체 변수 | `.` | `s1.name` |
| 구조체 포인터 | `->` | `p->name` (= `(*p).name`) |

### 11-7

```c
#include <stdio.h>
#include <string.h>

struct Student {
    char name[20];
    int age;
    int score;
};

int main() {
    struct Student s1;

    scanf("%s %d %d", s1.name, &s1.age, &s1.score);

    printf("이름: %s\n", s1.name);
    printf("나이: %d\n", s1.age);
    printf("점수: %d\n", s1.score);

    return 0;
}
```

    char name[20];
    int age;
    int score;

    struct Student students[3] = {

        {"김철수", 22, 95},
        {"이영희", 21, 75}
    };
    char search[20];
    int i, found = 0;

    scanf("%s", search);

        if (strcmp(students[i].name, search) == 0) {
            printf("찾았습니다!\n");

            found = 1;
            break;
        }
    }

    if (!found)
        printf("찾을 수 없습니다.\n");

}

# 12주차 정답

| 번호 | 정답 |
|------|------|
| 12-1 | ② `FILE *fp;` |
| 12-2 | ② 쓰기 (기존 파일 덮어쓰기) |
| 12-3 | ② `fclose(fp)` |
| 12-4 | ③ `NULL` |

| 12-6 | ② `fgets` |

### 파일 모드 정리

| 모드 | 의미 |
|------|------|
| `"r"` | 읽기 |
| `"w"` | 쓰기 (덮어쓰기) |
| `"a"` | 추가 (append) |
| `"r+"` | 읽기/쓰기 |

    return 0;
}

### 12-13 (성적 관리)

```c
#include <stdio.h>
#include <string.h>

struct Student {
    char name[20];
    int score;
};

char getGrade(int score) {
    if (score >= 90) return 'A';
    if (score >= 80) return 'B';
    if (score >= 70) return 'C';
    if (score >= 60) return 'D';
    return 'F';
}

int main() {
    struct Student students[3];
    FILE *fp;
    int i, sum = 0, max, min;

    for (i = 0; i < 3; i++)
        scanf("%s %d", students[i].name, &students[i].score);

    max = min = students[0].score;
    for (i = 0; i < 3; i++) {
        sum += students[i].score;
        if (students[i].score > max) max = students[i].score;
        if (students[i].score < min) min = students[i].score;
    }

    printf("평균: %.2f\n", sum / 3.0);
    printf("최고: %d\n", max);
    printf("최저: %d\n", min);

    fp = fopen("result.txt", "w");
    if (fp != NULL) {
        fprintf(fp, "평균: %.2f\n", sum / 3.0);
        fprintf(fp, "최고: %d\n", max);
        fprintf(fp, "최저: %d\n", min);
        for (i = 0; i < 3; i++) {
            fprintf(fp, "%s %d %c\n",
                    students[i].name, students[i].score,
                    getGrade(students[i].score));
        }
        fclose(fp);
    }

    return 0;
}
```

### 12-14 (간단한 일기장)

```c
#include <stdio.h>
#include <string.h>

void writeDiary() {
    FILE *fp = fopen("diary.txt", "a");
    char buffer[500];

    if (fp == NULL) {
        printf("파일을 열 수 없습니다.\n");
        return;
    }

    printf("일기 내용 입력: ");
    getchar();
    fgets(buffer, 500, stdin);
    fputs(buffer, fp);

    fclose(fp);
    printf("저장 완료!\n");
}

void readDiary() {
    FILE *fp = fopen("diary.txt", "r");
    char buffer[500];

    if (fp == NULL) {
        printf("일기 파일이 없습니다.\n");
        return;
    }

    printf("=== 일기 ===\n");
    while (fgets(buffer, 500, fp) != NULL)
        printf("%s", buffer);

    fclose(fp);
}

int main() {
    int menu;

    while (1) {
        printf("\n1. 일기 쓰기  2. 일기 읽기  3. 종료\n");
        printf("선택: ");
        scanf("%d", &menu);

        if (menu == 1) writeDiary();
        else if (menu == 2) readDiary();
        else if (menu == 3) break;
        else printf("잘못된 선택\n");
    }

    return 0;
}
```
