예제 1: 할일 목록 (Todo List)

배열 메서드와 상태 관리 패턴

// React에서 사용하는 방식 const [todos, setTodos] = useState([]); // 추가 const addTodo = (text) => { setTodos([...todos, { id: Date.now(), text, done: false }]); }; // 완료 처리 const toggleTodo = (id) => { setTodos(todos.map(todo => todo.id === id ? { ...todo, done: !todo.done } : todo )); }; // 삭제 const deleteTodo = (id) => { setTodos(todos.filter(todo => todo.id !== id)); };

    예제 2: 사용자 목록 필터링

    filter와 map을 활용한 검색

    const users = [ { id: 1, name: "홍길동", age: 25 }, { id: 2, name: "김철수", age: 30 }, { id: 3, name: "이영희", age: 28 } ]; // 나이가 27 이상인 사용자만 필터링 const filtered = users .filter(user => user.age >= 27) .map(user => `${user.name} (${user.age}세)`);
